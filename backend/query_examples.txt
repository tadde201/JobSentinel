# Example 5: Jobs by company
# SELECT c.name, COUNT(j.job_id) as total_jobs 
# FROM companies c 
# LEFT JOIN jobs j ON c.company_id = j.company_id 
# GROUP BY c.company_id, c.name;

# Example 6: User registrations
# SELECT user_id, email, full_name, created_at FROM users ORDER BY created_at DESC;

# Example 7: Applications by user
# SELECT u.full_name, COUNT(a.application_id) as applications 
# FROM users u 
# LEFT JOIN applications a ON u.user_id = a.user_id 
# GROUP BY u.user_id, u.full_name;
